using System;
using System.IO;
using System.Linq;
using System.Text;
using System.Collections.Generic;
using System.Security.Cryptography;
using System.Text.RegularExpressions;
using Microsoft.Build.Framework;
using Microsoft.Build.Utilities;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

// Build-time source asset import only. Does not load or execute upstream JS.
public sealed class ImportRunMimeRegistry : Task
{
    [Required] public string SourceRoot { get; set; }
    [Required] public string Output { get; set; }

    public override bool Execute()
    {
        var types = Path.Combine(SourceRoot, "mime-types");
        var db = Path.Combine(SourceRoot, "mime-db");
        if ((string)JObject.Parse(File.ReadAllText(Path.Combine(types, "package.json")))["version"] != "3.0.2"
            || (string)JObject.Parse(File.ReadAllText(Path.Combine(db, "package.json")))["version"] != "1.54.0")
            throw new InvalidOperationException("Review the MIME registry/scoring algorithm before changing pinned versions.");
        RequireHash(Path.Combine(types, "index.js"), "7de0cc894fd1d45cc9e4d0b5ad76d00aca6cb04966bed947bdfbf7d710c61c06");
        RequireHash(Path.Combine(types, "mimeScore.js"), "4f5d57657dc1180bd282485edf8bc95ebaecd405bdb405f4c279ab8d51374e43");
        RequireHash(Path.Combine(db, "db.json"), "96b8a5746867c832ab56743c05e46e73c9facb04879677df0b356f20496cb6cd");
        var registry = JObject.Parse(File.ReadAllText(Path.Combine(db, "db.json")));
        var map = new Dictionary<string, string>(StringComparer.Ordinal);
        foreach (var property in registry.Properties())
        {
            var extensions = property.Value["extensions"] as JArray;
            if (extensions == null) continue;
            foreach (var extension in extensions.Values<string>())
            {
                string previous;
                var oldScore = map.TryGetValue(extension, out previous) ? Score(previous, (string)registry[previous]["source"]) : 0;
                if (oldScore <= Score(property.Name, (string)property.Value["source"])) map[extension] = property.Name;
            }
        }
        var text = new StringBuilder("// <auto-generated> ImportSourceRunMimeRegistry; do not edit. </auto-generated>\n");
        text.AppendLine("// mime-types 3.0.2 + mime-db 1.54.0. Source index.js _preferredType and mimeScore.js.");
        foreach (var input in new[] { Path.Combine(types, "index.js"), Path.Combine(types, "mimeScore.js"), Path.Combine(db, "db.json") })
        {
            using (var hash = SHA256.Create()) text.AppendLine("// " + Path.GetFileName(input) + " SHA-256 " + BitConverter.ToString(hash.ComputeHash(File.ReadAllBytes(input))).Replace("-", "").ToLowerInvariant());
        }
        foreach (var license in new[] { Path.Combine(types, "LICENSE"), Path.Combine(db, "LICENSE") })
            foreach (var line in File.ReadAllLines(license)) text.AppendLine("// " + line);
        text.AppendLine("namespace OpenCode.Cli.Commands.Run;");
        text.AppendLine("internal static partial class RunMimeRegistry");
        text.AppendLine("{");
        text.AppendLine("    private static readonly System.Collections.Generic.Dictionary<string, string> Types = new(System.StringComparer.Ordinal)");
        text.AppendLine("    {");
        foreach (var item in map.OrderBy(item => item.Key, StringComparer.Ordinal))
            text.AppendLine("        [" + JsonConvert.SerializeObject(item.Key) + "] = " + JsonConvert.SerializeObject(item.Value) + ",");
        text.AppendLine("    };");
        text.AppendLine("}");
        File.WriteAllText(Output, text.ToString(), new UTF8Encoding(false));
        Log.LogMessage(MessageImportance.High, "Imported {0} reviewed MIME extensions into {1}", map.Count, Output);
        return true;
    }

    private static double Score(string mime, string source)
    {
        if (mime == "application/octet-stream") return 0;
        var parts = mime.Split('/');
        var facet = Regex.Replace(parts[1], @"(\.|x-).*", "$1");
        var facetScore = facet == "prs." ? 100 : facet == "x-" ? 200 : facet == "x." ? 300 : facet == "vnd." ? 400 : 900;
        var sourceScore = source == "nginx" ? 10 : source == "apache" ? 20 : source == "iana" ? 40 : 30;
        var typeScore = parts[0] == "application" ? 1 : parts[0] == "font" || parts[0] == "audio" ? 2 : parts[0] == "video" ? 3 : 0;
        return facetScore + sourceScore + typeScore + (1 - mime.Length / 100d);
    }

    private static void RequireHash(string file, string expected)
    {
        using (var hash = SHA256.Create())
            if (BitConverter.ToString(hash.ComputeHash(File.ReadAllBytes(file))).Replace("-", "").ToLowerInvariant() != expected)
                throw new InvalidOperationException("MIME source changed; review before regeneration: " + Path.GetFileName(file));
    }
}
